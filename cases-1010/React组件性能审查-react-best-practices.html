① 审查对象

一个典型列表页组件：数据获取 + 渲染 + 交互（类业务组件审查样本）
function ProductList({ categoryId }) {
  const [items, setItems] = useState([])
  const [page, setPage] = useState(1)

  useEffect(() => { fetchItems(categoryId, page).then(setItems) }, [categoryId])
  // 注：page 未进依赖，翻页不触发重新获取
  return items.map(it => <Row item={it} key={it.id} />)
}

严重度规则问题建议

③ 修复后示例

const Row = memo(function Row({ item }) { ... })

function ProductList({ categoryId }) {
  const [state, setState] = useReducer(reducer, init)

  useEffect(() => {
    const ctrl = new AbortController()
    fetchItems(categoryId, state.page, { signal: ctrl.signal })
      .then(items => setState({ type: 'loaded', items }))
    return () => ctrl.abort()
  }, [categoryId, state.page])

  return state.items.map(it => <Row item={it} key={it.id} />)
}
